'border-red-500' : 'border-zinc-700'} rounded p-3 text-white focus:outline-none focus:border-red-500`} value={formData.age} onChange={e => setFormData({...formData, age: e.target.value})} />

'border-red-500' : 'border-zinc-700'} rounded p-3 text-white focus:outline-none focus:border-red-500`} value={formData.parentName} onChange={e => setFormData({...formData, parentName: e.target.value})} />
                    {errors.parentName && <p className="text-red-500 text-xs mt-1">{errors.parentName}</p>}
                  </div>
                </div>

                  <div>

                    {errors.phone && <p className="text-red-500 text-xs mt-1">{errors.phone}</p>}

                  <div className="md:col-span-2">
                    <label className="block text-sm font-medium text-zinc-400 mb-1">Program *</label>
                    <select className="w-full bg-zinc-950 border border-zinc-700 rounded p-3 text-white focus:outline-none focus:border-red-500" value={formData.program} onChange={e => setFormData({...formData, program: e.target.value})}>
                      {PROGRAMS.map(p => <option key={p.id} value={p.title}>{p.title}</option>)}
                    </select>
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-zinc-400 mb-1">Preferred Date *</label>
                    <input type="date" className={`w-full bg-zinc-950 border ${errors.date ? 'border-red-500' : 'border-zinc-700'} rounded p-3 text-white focus:outline-none focus:border-red-500`} value={formData.date} onChange={e => setFormData({...formData, date: e.target.value})} />
                    {errors.date && <p className="text-red-500 text-xs mt-1">{errors.date}</p>}
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-zinc-400 mb-1">Preferred Time *</label>
                    <select className="w-full bg-zinc-950 border border-zinc-700 rounded p-3 text-white focus:outline-none focus:border-red-500" value={formData.time} onChange={e => setFormData({...formData, time: e.target.value})}>
                      {['4:00 PM', '4:30 PM', '5:00 PM', '5:30 PM', '6:00 PM', '6:30 PM', '7:00 PM', '7:30 PM'].map(t => <option key={t} value={t}>{t}</option>)}
                    </select>
                  </div>
                </div>
              </div>

              {/* Additional Info */}
              <div className="pt-6 border-t border-zinc-800">
                <div className="space-y-6">
                  <div>
                    <label className="block text-sm font-medium text-zinc-400 mb-1">Previous Experience</label>
                    <select className="w-full bg-zinc-950 border border-zinc-700 rounded p-3 text-white focus:outline-none focus:border-red-500" value={formData.experience} onChange={e => setFormData({...formData, experience: e.target.value})}>
                      <option value="None">None - Brand New</option>
                      <option value="Less than 1 year">Less than 1 year</option>
                      <option value="1-3 years">1-3 years</option>
                      <option value="3+ years">3+ years</option>
                    </select>
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-zinc-400 mb-1">Any questions or injuries we should know about?</label>
                    <textarea rows="3" className="w-full bg-zinc-950 border border-zinc-700 rounded p-3 text-white focus:outline-none focus:border-red-500" value={formData.message} onChange={e => setFormData({...formData, message: e.target.value})}></textarea>
                  </div>
                </div>
              </div>

              <div className="pt-8">
                <button type="submit" className="w-full bg-red-600 hover:bg-red-700 text-white py-4 rounded-lg font-black text-xl transition-all shadow-lg shadow-red-900/20 active:scale-[0.98]">
                  BOOK MY FREE TRIAL
                </button>
                <p className="text-center text-zinc-500 text-xs mt-4">
                  By booking, you agree to our privacy policy. No credit card required.
                </p>
              </div>
            </form>
          </div>
        </div>
      </div>
    );
  };

  const ConfirmationView = () => {
    if (!lastBooking) return null;
    
    const handleCalendar = () => {
      // Mock calendar integration
      const el = document.createElement('div');
      el.className = 'fixed inset-0 bg-black/80 flex items-center justify-center z-50 p-4';
      el.innerHTML = `
        <div class="bg-zinc-900 p-8 rounded-xl border border-zinc-700 max-w-sm w-full text-center">
          <div class="w-16 h-16 bg-blue-500/20 text-blue-500 rounded-full flex items-center justify-center mx-auto mb-4">
            <svg xmlns="http://www.w3.org/2000/svg" width="32" height="32" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect><line x1="16" y1="2" x2="16" y2="6"></line><line x1="8" y1="2" x2="8" y2="6"></line><line x1="3" y1="10" x2="21" y2="10"></line></svg>
          </div>
          <h3 class="text-xl font-bold text-white mb-2">Event Added!</h3>
          <p class="text-zinc-400 mb-6">Demo note: In a real app, this would download an .ics file or open Google Calendar.</p>
          <button class="bg-zinc-800 text-white px-6 py-2 rounded font-bold w-full" onclick="this.parentElement.parentElement.remove()">Close</button>
        </div>
      `;
      document.body.appendChild(el);
    };

    return (
      <div className="min-h-screen bg-zinc-950 flex flex-col items-center justify-center p-4 animate-in zoom-in-95 duration-500">
        <div className="bg-zinc-900 border border-zinc-800 rounded-2xl p-8 md:p-12 max-w-lg w-full text-center shadow-2xl">
          <div className="w-20 h-20 bg-green-500/10 rounded-full flex items-center justify-center mx-auto mb-6">
            <CheckCircle className="text-green-500" size={48} />
          </div>
          
          <h2 className="text-3xl font-black text-white mb-2 uppercase tracking-tight">Your Free Trial is Booked!</h2>
          <p className="text-zinc-400 text-lg mb-8">We look forward to seeing you on the mat.</p>
          
          <div className="bg-zinc-950 border border-zinc-800 rounded-xl p-6 text-left mb-8 space-y-4">
            <div>
              <p className="text-zinc-500 text-xs uppercase font-bold tracking-wider">Student</p>
              <p className="text-white font-medium text-lg">{lastBooking.name}</p>
            </div>
            <div>
              <p className="text-zinc-500 text-xs uppercase font-bold tracking-wider">Program</p>
              <p className="text-white font-medium text-lg">{lastBooking.program}</p>
            </div>
            <div className="flex gap-8">
              <div>
                <p className="text-zinc-500 text-xs uppercase font-bold tracking-wider">Date</p>
                <p className="text-white font-medium text-lg">{new Date(lastBooking.date).toLocaleDateString(undefined, { weekday: 'long', month: 'long', day: 'numeric'})}</p>
              </div>
              <div>
                <p className="text-zinc-500 text-xs uppercase font-bold tracking-wider">Time</p>
                <p className="text-white font-medium text-lg">{lastBooking.time}</p>
              </div>
            </div>
            <div className="pt-4 border-t border-zinc-800">
              <p className="text-zinc-500 text-xs uppercase font-bold tracking-wider">Location</p>
              <p className="text-white font-medium">Zero Gravity BJJ Academy<br/><span className="text-zinc-400 font-normal">Brooklyn, New York</span></p>
            </div>
          </div>
          
          <div className="space-y-4">
            <button onClick={handleCalendar} className="w-full bg-blue-600 hover:bg-blue-700 text-white py-4 rounded-lg font-bold text-lg transition-colors flex items-center justify-center gap-2">
              <Calendar size={20} /> ADD TO CALENDAR
            </button>
            <button onClick={() => navigateTo('home')} className="w-full bg-zinc-800 hover:bg-zinc-700 text-white py-4 rounded-lg font-bold text-lg transition-colors">
              BACK TO HOME
            </button>
          </div>
        </div>
      </div>
    );
  };

  const AdminView = () => (
    <div className="min-h-screen bg-zinc-950 p-4 sm:p-8 animate-in fade-in">
      <div className="max-w-7xl mx-auto">
        <div className="flex justify-between items-center mb-8">
          <div>
            <h1 className="text-3xl font-black text-white">Admin Dashboard</h1>
            <p className="text-zinc-400">Demo Data Management</p>
          </div>
          <button onClick={() => navigateTo('home')} className="bg-zinc-800 text-white px-4 py-2 rounded text-sm font-medium hover:bg-zinc-700">
            Exit Admin
          </button>
        </div>

        <div className="bg-zinc-900 border border-zinc-800 rounded-xl overflow-x-auto">
          {bookings.length === 0 ? (
            <div className="p-12 text-center text-zinc-500">
              No trial bookings found in localStorage.
            </div>
          ) : (
            <table className="w-full text-left border-collapse min-w-[800px]">
              <thead>
                <tr className="bg-zinc-950 text-zinc-400 text-sm uppercase tracking-wider">
                  <th className="p-4 border-b border-zinc-800 font-bold">Student</th>
                  <th className="p-4 border-b border-zinc-800 font-bold">Contact</th>
                  <th className="p-4 border-b border-zinc-800 font-bold">Class Details</th>
                  <th className="p-4 border-b border-zinc-800 font-bold">Status</th>
                  <th className="p-4 border-b border-zinc-800 font-bold text-right">Actions</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-zinc-800 text-zinc-300">
                {bookings.map(b => (
                  <tr key={b.id} className="hover:bg-zinc-800/50 transition-colors">
                    <td className="p-4">
                      <div className="font-bold text-white">{b.name}</div>
                      <div className="text-xs text-zinc-500">Age: {b.age} {b.parentName && `(Parent: ${b.parentName})`}</div>
                    </td>
                    <td className="p-4">
                      <div className="text-sm">{b.phone}</div>
                      <div className="text-xs text-zinc-500">{b.email}</div>
                    </td>
                    <td className="p-4">
                      <div className="text-sm font-medium text-white">{b.program}</div>
                      <div className="text-xs text-zinc-500">{b.date} at {b.time}</div>
                    </td>
                    <td className="p-4">
                      <select 
                        value={b.status} 
                        onChange={(e) => updateBookingStatus(b.id, e.target.value)}
                        className={`text-xs font-bold px-2 py-1 rounded border outline-none ${
                          b.status === 'New' ? 'bg-blue-900/30 text-blue-400 border-blue-900' :
                          b.status === 'Contacted' ? 'bg-yellow-900/30 text-yellow-400 border-yellow-900' :
                          'bg-green-900/30 text-green-400 border-green-900'
                        }`}
                      >
                        <option value="New">New</option>
                        <option value="Contacted">Contacted</option>
                        <option value="Confirmed">Confirmed</option>
                      </select>
                    </td>
                    <td className="p-4 text-right">
                      <button onClick={() => deleteBooking(b.id)} className="text-zinc-600 hover:text-red-500 transition-colors p-2" title="Delete Booking">
                        <Trash2 size={18} />
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </div>
      </div>
    </div>
  );

  return (
    <div className="min-h-screen flex flex-col bg-zinc-950 font-sans selection:bg-red-600 selection:text-white">
      {currentView !== 'admin' && <Navbar />}
      
      <main className="flex-grow flex flex-col">
        {currentView === 'home' && <HomeView />}
        {currentView === 'booking' && <BookingView />}
        {currentView === 'confirmation' && <ConfirmationView />}
        {currentView === 'admin' && <AdminView />}
      </main>

      {currentView !== 'admin' && <Footer />}
    </div>
  );
}
